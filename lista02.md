// a) Escreva um programa que peça ao usuário para digitar dois números e exiba a soma deles.
Algoritmo "Exercicio_A"
Var
   num1, num2, soma: real
Inicio
   Escreva("Digite o primeiro número: ")
   Leia(num1)
   Escreva("Digite o segundo número: ")
   Leia(num2)
   soma <- num1 + num2
   Escreval("A soma é: ", soma)
Fimalgoritmo
